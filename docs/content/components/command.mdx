## Command Dialog (Palette)

Trigger a global quick switcher modal using `<CommandDialog />`.

<ComponentPreview
  title="Modal Command Palette"
  description="Accessible dialog modal with auto-focused search and keyboard shortcut."
  code={'import { useState } from "react";\nimport {\n  CommandDialog,\n  CommandInput,\n  CommandList,\n  CommandEmpty,\n  CommandGroup,\n  CommandItem,\n  CommandShortcut,\n  Button,\n  Kbd,\n} from "@galaui/react";\nimport { Sparkles, Layers, FileText } from "lucide-react";\n\nexport default function Example() {\n  const [open, setOpen] = useState(false);\n\n  return (\n    <div>\n      <Button\n        variant="outline"\n        className="text-sm text-muted-foreground flex items-center gap-3 px-3 py-2"\n        onClick={() => setOpen(true)}\n      >\n        <span>Search application...</span>\n        <Kbd>⌘K</Kbd>\n      </Button>\n\n      <CommandDialog open={open} onOpenChange={setOpen}>\n        <CommandInput placeholder="Type a command or search..." />\n        <CommandList>\n          <CommandEmpty>No matching commands.</CommandEmpty>\n          <CommandGroup heading="Actions">\n            <CommandItem onSelect={() => setOpen(false)}>\n              <Sparkles className="mr-2 h-4 w-4" />\n              <span>Generate Component</span>\n            </CommandItem>\n            <CommandItem onSelect={() => setOpen(false)}>\n              <Layers className="mr-2 h-4 w-4" />\n              <span>Explore Design Tokens</span>\n            </CommandItem>\n            <CommandItem onSelect={() => setOpen(false)}>\n              <FileText className="mr-2 h-4 w-4" />\n              <span>View Documentation</span>\n              <CommandShortcut>⌘D</CommandShortcut>\n            </CommandItem>\n          </CommandGroup>\n        </CommandList>\n      </CommandDialog>\n    </div>\n  );\n}'}
>
  {(() => {
    const [open, setOpen] = useState(false);
    return (
      <div>
        <Button
          variant="outline"
          className="text-sm text-muted-foreground flex items-center gap-4 px-3.5 py-2 h-9"
          onClick={() => setOpen(true)}
        >
          <Search className="w-3.5 h-3.5" />
          <span>Open Command Palette...</span>
          <Kbd className="text-[10px] ml-2">⌘K</Kbd>
        </Button>

        <CommandDialog open={open} onOpenChange={setOpen}>
          <CommandInput placeholder="Type a command or search..." />
          <CommandList>
            <CommandEmpty>No matching commands.</CommandEmpty>
            <CommandGroup heading="Quick Actions">
              <CommandItem onSelect={() => setOpen(false)}>
                <Sparkles className="mr-2 h-4 w-4 text-primary" />
                <span>Generate New Component</span>
              </CommandItem>
              <CommandItem onSelect={() => setOpen(false)}>
                <Layers className="mr-2 h-4 w-4 text-muted-foreground" />
                <span>Explore Design Tokens</span>
              </CommandItem>
              <CommandItem onSelect={() => setOpen(false)}>
                <FileText className="mr-2 h-4 w-4 text-muted-foreground" />
                <span>Read Getting Started</span>
                <CommandShortcut>⌘G</CommandShortcut>
              </CommandItem>
            </CommandGroup>
          </CommandList>
        </CommandDialog>
      </div>
    );
  })()}
</ComponentPreview>
