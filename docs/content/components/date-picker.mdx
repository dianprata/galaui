import * as React from "react";
import {
  DatePicker,
  DateRangePicker,
  defaultPresets,
  Field,
  FieldLabel,
  FieldDescription,
  Badge,
} from "@/index";
import { ComponentPreview } from "@docs/components/ComponentPreview";
import { CodeBlock } from "@docs/components/CodeBlock";
import { ApiReference } from "@docs/components/ApiReference";

export function SingleDatePickerDemo() {
  const [date, setDate] = React.useState(() => {
    const d = new Date();
    d.setDate(d.getDate() + 4);
    return d;
  });

  return (
    <div className="w-full max-w-xs flex flex-col items-center gap-2.5">
      <DatePicker
        value={date}
        onChange={setDate}
        placeholder="Choose a date..."
        clearable
        className="w-full"
      />
      <span className="text-xs font-mono text-muted-foreground bg-muted/40 px-2.5 py-1 rounded-md border border-border">
        Selected: <strong className="text-foreground">{date ? date.toLocaleDateString("en-US", { month: "short", day: "numeric", year: "numeric" }) : "None"}</strong>
      </span>
    </div>
  );
}

export function DateRangePickerDemo() {
  const [range, setRange] = React.useState(() => {
    const from = new Date();
    from.setDate(from.getDate() + 2);
    const to = new Date();
    to.setDate(from.getDate() + 10);
    return { from, to };
  });

  return (
    <div className="w-full max-w-sm flex flex-col items-center gap-2.5">
      <DateRangePicker
        value={range}
        onChange={setRange}
        presets={defaultPresets}
        clearable
        className="w-full"
      />
      <span className="text-xs font-mono text-muted-foreground bg-muted/40 px-2.5 py-1 rounded-md border border-border">
        Range: <strong className="text-foreground">{range?.from ? range.from.toLocaleDateString("en-US", { month: "short", day: "numeric" }) : "..."} – {range?.to ? range.to.toLocaleDateString("en-US", { month: "short", day: "numeric", year: "numeric" }) : "..."}</strong>
      </span>
    </div>
  );
}

export function FormIntegrationDemo() {
  const [birthDate, setBirthDate] = React.useState(() => new Date(2000, 4, 15));

  return (
    <div className="w-full max-w-xs flex justify-center">
      <Field className="w-full space-y-1.5">
        <FieldLabel>Date of Birth</FieldLabel>
        <DatePicker
          value={birthDate}
          onChange={setBirthDate}
          placeholder="Select your birth date"
          className="w-full"
          clearable
        />
        <FieldDescription>Used to calculate age and verify eligibility.</FieldDescription>
      </Field>
    </div>
  );
}
