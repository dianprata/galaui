# Date Picker

A date and range picker component combining an interactive input trigger button, popover positioning, and the GalaUI Calendar.

## Single Date Picker

<ComponentPreview

## Date Range Picker with Presets

<ComponentPreview
  title="Date Range Picker with Presets"
  description="Select start and end dates with side-by-side dual calendars and quick range presets."
  code={'import * as React from "react";\nimport { DateRangePicker, defaultPresets, type DateRange } from "@galaui/react";\n\nexport default function Example() {\n  const [range, setRange] = React.useState<DateRange | undefined>(() => {\n    const from = new Date();\n    const to = new Date();\n    to.setDate(from.getDate() + 14);\n    return { from, to };\n  });\n\n  return (\n    <div className="w-full max-w-sm flex justify-center">\n      <DateRangePicker\n        value={range}\n        onChange={setRange}\n        presets={defaultPresets}\n        clearable\n      />\n    </div>\n  );\n}'}
>
  <div className="w-full max-w-sm flex justify-center">
    <DateRangePicker presets={defaultPresets} clearable />
  </div>
</ComponentPreview>
