# Fieldset

An accessible form grouping container powered by Base UI Fieldset, providing semantic grouping for related form controls with legend titles, descriptions, and automatic disabled cascading.

## Disabled Cascade

Setting `disabled` on `<Fieldset />` automatically cascades to disable all internal form controls.

<ComponentPreview
  title="Disabled State"
  description="Fieldset disabled state prevents interaction across all grouped controls."
  code={'import {\n  Fieldset,\n  FieldsetLegend,\n  FieldsetDescription,\n  Field,\n  FieldLabel,\n  Input,\n  Button,\n} from "@galaui/react";\n\nexport default function Example() {\n  return (\n    <div className="w-full max-w-md">\n      <Fieldset variant="bordered" disabled>\n        <FieldsetLegend>Billing Account (Locked)</FieldsetLegend>\n        <FieldsetDescription>Account modifications are currently restricted.</FieldsetDescription>\n        <div className="space-y-3 pt-1">\n          <Field>\n            <FieldLabel>Tax ID</FieldLabel>\n            <Input value="TAX-9821-X" readOnly />\n          </Field>\n          <Button size="sm">Update Payment Method</Button>\n        </div>\n      </Fieldset>\n    </div>\n  );\n}'}
>
  <div className="w-full max-w-md">
    <Fieldset variant="bordered" disabled>
      <FieldsetLegend>Billing Account (Locked)</FieldsetLegend>
      <FieldsetDescription>Account modifications are currently restricted.</FieldsetDescription>
      <div className="space-y-3 pt-1">
        <Field>
          <FieldLabel>Tax ID</FieldLabel>
          <Input value="TAX-9821-X" readOnly />
        </Field>
        <Button size="sm">Update Payment Method</Button>
      </div>
    </Fieldset>
  </div>
</ComponentPreview>
