## Horizontal Stepper

Interactive multi-step wizard showing completed, current, and upcoming stages.

    const [step, setStep] = useState(2);
    return (
      <div className="w-full max-w-xl space-y-6">
        <Stepper value={step} onValueChange={setStep} clickable>
          <StepItem step={1}>
            <StepIndicator>1</StepIndicator>
            <StepContent>
              <StepTitle>Account</StepTitle>
              <StepDescription>Login details</StepDescription>
            </StepContent>
          </StepItem>
          <StepSeparator />
          <StepItem step={2}>

        <div className="flex justify-between pt-2">
          <Button
            variant="outline"
            size="sm"
            disabled={step <= 1}
            onClick={() => setStep((s) => s - 1)}

            Previous

          <Button
            size="sm"
            disabled={step >= 3}
            onClick={() => setStep((s) => s + 1)}
          >
            Next Step

        </div>
      </div>
