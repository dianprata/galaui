# Label

Renders an accessible label associated with form controls, styled with GalaUI typography tokens and peer-disabled states.

<div className="flex flex-wrap items-center gap-2 my-4">
  <Badge variant="outline" className="font-mono text-[11px]">Semantic HTML</Badge>
  <Badge variant="secondary" className="font-mono text-[11px]">Peer Accessibility</Badge>
</div>

## Import

<CodeBlock language="tsx" code={'import { Label } from "@galaui/react";'} />

## Example

<ComponentPreview
  title="Form Input with Label"
  description="Accessible text field with semantic label binding."
  code={'import { Label, Input } from "@galaui/react";\n\nexport default function Example() {\n  return (\n    <div className="w-full max-w-sm space-y-2">\n      <Label htmlFor="email">Email Address</Label>\n      <Input id="email" type="email" placeholder="name@company.com" />\n    </div>\n  );\n}'}
>
  <div className="w-full max-w-sm space-y-2">
    <Label htmlFor="email-preview">Email Address</Label>
    <Input id="email-preview" type="email" placeholder="name@company.com" />
  </div>
</ComponentPreview>
