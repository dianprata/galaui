export function SingleDateDemo() {
  const [date, setDate] = React.useState(() => {
    const d = new Date();
    d.setDate(d.getDate() + 4);
    return d;
  });

  const today = new Date();

  return (
    <div className="flex flex-col items-center gap-3 w-full">
      <Calendar
        mode="single"
        selected={date}
        onSelect={setDate}
      />
      <div className="flex flex-wrap items-center justify-center gap-3 text-xs font-mono text-muted-foreground bg-muted/40 px-3 py-1.5 rounded-lg border border-border">
        <span>Selected: <strong className="text-foreground">{date ? date.toLocaleDateString("en-US", { month: "short", day: "numeric", year: "numeric" }) : "None"}</strong></span>
        <span>•</span>
        <span>Today: <span className="text-primary font-semibold underline underline-offset-4 decoration-primary/40">{today.toLocaleDateString("en-US", { month: "short", day: "numeric" })}</span></span>
      </div>
    </div>
  );
}

export function DateRangeDemo() {
  const [range, setRange] = React.useState(() => {
    const from = new Date();
    from.setDate(from.getDate() + 2);
    const to = new Date();
    to.setDate(from.getDate() + 7);
    return { from, to };
  });

  return (
    <div className="flex flex-col items-center gap-3 w-full">
      <Calendar
        mode="range"
        selected={range}
        onSelect={setRange}
      />
      <div className="flex items-center gap-2 text-xs font-mono text-muted-foreground bg-muted/40 px-3 py-1.5 rounded-lg border border-border">
        <span>Selected Range:</span>
        <strong className="text-foreground">
          {range?.from ? range.from.toLocaleDateString("en-US", { month: "short", day: "numeric", year: "numeric" }) : "Start date"}
          {" – "}
          {range?.to ? range.to.toLocaleDateString("en-US", { month: "short", day: "numeric", year: "numeric" }) : "End date"}
        </strong>
      </div>
    </div>
  );
}

export function MultiMonthDemo() {
  const [range, setRange] = React.useState(() => {
    const from = new Date();
    const to = new Date();
    to.setDate(from.getDate() + 14);
    return { from, to };
  });

  return (
    <div className="flex flex-col items-center gap-3 w-full overflow-x-auto py-2">
      <Calendar
        mode="range"
        numberOfMonths={2}
        selected={range}
        onSelect={setRange}
      />
    </div>
  );
}

## Single Date Selection

Di contoh ini, tanggal yang dipilih sengaja diatur **4 hari ke depan** agar kamu dapat membedakan dengan jelas antara indikator hari ini (**Today** bergaris bawah dengan teks primer) dan tanggal yang sedang dipilih (**Selected** dengan background solid primer).

## Date Range Selection

Di mode rentang (*range*), pengguna dapat memilih tanggal awal (*from*) dan tanggal akhir (*to*). Rentang tanggal terpilih ditandai dengan continuous background highlight, dan status tanggal yang sedang terpilih ditampilkan di bawah kalender.

<ComponentPreview

  description="Select a start and end date with continuous range fill and hover preview."

>

</ComponentPreview>
