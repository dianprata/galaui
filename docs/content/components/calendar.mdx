# Calendar

A clean, accessible date and range calendar component featuring monthly grid views, date range selection with hover highlights, multi-month layouts, and localized date formatting.

## Single Date Selection

<ComponentPreview
  title="Single Date"
  description="Calendar supporting single date selection with today indicator."
  code={'import * as React from "react";\nimport { Calendar } from "@galaui/react";\n\nexport default function Example() {\n  const [date, setDate] = React.useState<Date | undefined>(new Date());\n\n  return (\n    <div className="flex flex-col items-center gap-3">\n      <Calendar\n        mode="single"\n        selected={date}\n        onSelect={setDate}\n      />\n      <p className="text-xs text-muted-foreground font-mono">\n        Selected: {date ? date.toLocaleDateString("en-US", { dateStyle: "long" }) : "None"}\n      </p>\n    </div>\n  );\n}'}
>
  <div className="flex flex-col items-center gap-3">
    <Calendar mode="single" />
  </div>
</ComponentPreview>

## Date Range Selection

<ComponentPreview
  title="Date Range"
  description="Select a start and end date with continuous range fill and hover preview."
  code={'import * as React from "react";\nimport { Calendar, type DateRange } from "@galaui/react";\n\nexport default function Example() {\n  const [range, setRange] = React.useState<DateRange | undefined>(() => {\n    const from = new Date();\n    const to = new Date();\n    to.setDate(from.getDate() + 7);\n    return { from, to };\n  });\n\n  return (\n    <div className="flex flex-col items-center gap-3">\n      <Calendar\n        mode="range"\n        selected={range}\n        onSelect={setRange}\n      />\n    </div>\n  );\n}'}
>
  <div className="flex flex-col items-center gap-3">
    <Calendar mode="range" />
  </div>
</ComponentPreview>

## Disabled Dates & Constraints

<ComponentPreview
  title="Disabled Constraints"
  description="Disable past dates and weekend days with minDate and custom matchers."
